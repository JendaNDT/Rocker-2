cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(Rocker2 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8 /bigobj /MP2)
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.5 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_binary_data(Fonts SOURCES assets/fonts/Inter.ttf assets/fonts/InterSemiBold.ttf assets/fonts/Mono.ttf)
add_library(Studio STATIC src/engine/Audio.cpp src/engine/Devices.cpp src/storage/Settings.cpp src/ui/Studio.cpp)
target_include_directories(Studio PUBLIC src "${JUCE_MODULES_DIR}")
target_compile_definitions(Studio PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 $<$<PLATFORM_ID:Windows>:JUCE_ASIO=1>)
target_link_libraries(Studio PRIVATE Fonts juce::juce_audio_devices juce::juce_gui_basics
    PUBLIC juce::juce_recommended_config_flags)
juce_add_gui_app(Rocker2 PRODUCT_NAME "Rocker 2" BUNDLE_ID "cz.rocker2.studio"
    MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Rocker 2 potřebuje vstup pro měření signálu z nástroje.")
target_sources(Rocker2 PRIVATE src/app/Main.cpp)
target_link_libraries(Rocker2 PRIVATE Studio)
include(CTest)
if(BUILD_TESTING)
    set(CATCH_CONFIG_WINDOWS_CRTDBG OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.8.1 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(Catch2)
    juce_add_console_app(StudioTests PRODUCT_NAME StudioTests)
    target_sources(StudioTests PRIVATE tests/AudioTests.cpp tests/StudioTests.cpp)
    target_link_libraries(StudioTests PRIVATE Studio Catch2::Catch2WithMain)
    add_test(NAME E0 COMMAND StudioTests)
    set_tests_properties(E0 PROPERTIES ENVIRONMENT "ROCKER_SNAPSHOTS=${CMAKE_BINARY_DIR}/screenshots")
endif()
