cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(Rocker2 VERSION 0.7.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8 /bigobj /MP2)
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.5 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_binary_data(Fonts SOURCES assets/fonts/Inter.ttf assets/fonts/InterSemiBold.ttf assets/fonts/Mono.ttf)
add_library(Studio STATIC src/song/Arrangement.cpp src/song/Edit.cpp src/engine/LoadTrial.cpp src/song/GuitarSound.cpp src/storage/AtomicFile.cpp src/song/Song.cpp src/engine/Timeline.cpp src/dsp/Amp.cpp src/dsp/Pitch.cpp src/dsp/Tunings.cpp src/engine/Live.cpp src/engine/Audio.cpp src/engine/Devices.cpp src/storage/Settings.cpp src/ui/Studio.cpp src/ui/LivePages.cpp src/ui/ProjectPage.cpp src/ui/EditPage.cpp src/ui/SectionPage.cpp)
target_include_directories(Studio PUBLIC src "${JUCE_MODULES_DIR}")
target_compile_definitions(Studio PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 $<$<PLATFORM_ID:Windows>:JUCE_ASIO=1>)
target_link_libraries(Studio PRIVATE Fonts juce::juce_audio_formats juce::juce_audio_devices juce::juce_gui_basics
    PUBLIC juce::juce_recommended_config_flags)
juce_add_gui_app(Rocker2 PRODUCT_NAME "Rocker 2" BUNDLE_ID "cz.rocker2.studio"
    MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Rocker 2 potřebuje vstup pro měření signálu z nástroje.")
target_sources(Rocker2 PRIVATE src/app/Main.cpp)
target_compile_definitions(Rocker2 PRIVATE ROCKER_VERSION="${PROJECT_VERSION}")
target_link_libraries(Rocker2 PRIVATE Studio)
if(APPLE)
    target_link_libraries(Studio PRIVATE "-framework IOKit")
endif()
include(CTest)
if(BUILD_TESTING)
    set(CATCH_CONFIG_WINDOWS_CRTDBG OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.8.1 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(Catch2)
    juce_add_console_app(StudioTests PRODUCT_NAME StudioTests)
    target_sources(StudioTests PRIVATE tests/EditTests.cpp tests/GuitarTests.cpp tests/SongTests.cpp tests/AudioTests.cpp tests/LiveTests.cpp tests/StudioTests.cpp)
    target_link_libraries(StudioTests PRIVATE Studio Catch2::Catch2WithMain)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    juce_add_console_app(CrashWorker PRODUCT_NAME CrashWorker)
    target_sources(CrashWorker PRIVATE tests/CrashWorker.cpp)
    target_link_libraries(CrashWorker PRIVATE Studio)
    foreach(trial A08 A09)
        add_test(NAME ${trial} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/crash_trials.py" "$<TARGET_FILE:CrashWorker>" "${trial}")
        set_tests_properties(${trial} PROPERTIES TIMEOUT 900)
    endforeach()
    option(ROCKER_SANITIZERS "Build realtime and threading audit binaries" OFF)
    if(ROCKER_SANITIZERS)
        foreach(sanitizer realtime thread)
            set(target Audit_${sanitizer})
            juce_add_console_app(${target} PRODUCT_NAME ${target})
            target_sources(${target} PRIVATE tests/SongTests.cpp tests/GuitarTests.cpp src/engine/Audio.cpp src/engine/Live.cpp
                src/engine/Timeline.cpp src/dsp/Pitch.cpp src/dsp/Amp.cpp src/song/Song.cpp)
            target_compile_options(${target} PRIVATE -O1 -g -fsanitize=${sanitizer} -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=${sanitizer})
            target_link_libraries(${target} PRIVATE Studio Catch2::Catch2WithMain)
            if(sanitizer STREQUAL "realtime")
                target_compile_definitions(${target} PRIVATE ROCKER_RT_AUDIT=1)
            endif()
            add_test(NAME A15_${sanitizer} COMMAND ${target} "[realtime]")
        endforeach()
    endif()
    add_test(NAME StudioSuite COMMAND StudioTests)
    set_tests_properties(StudioSuite PROPERTIES ENVIRONMENT "ROCKER_SNAPSHOTS=${CMAKE_BINARY_DIR}/screenshots")
endif()
